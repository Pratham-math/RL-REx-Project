def is_reachable(n, m, horizontal, vertical):
    # Check horizontal directions
    for i in range(n):
        if horizontal[i] == '<' and any(horizontal[j] == '>' for j in range(i + 1, n)):
            return "NO"
        if horizontal[i] == '>' and any(horizontal[j] == '<' for j in range(i + 1, n)):
            return "NO"
    
    # Check vertical directions

            return "NO"

            return "NO"
    
    return "YES"

def can_reach_all(n, m, horizontal, vertical):
    # Check horizontal streets
    for i in range(n):
        if horizontal[i] == '<':
            if i > 0 and horizontal[i-1] != '>':
                return False
        else:
            if i > 0 and horizontal[i-1] != '<':
                return False
    
    # Check vertical streets
    for i in range(m):
        if vertical[i] == '^':
            if i > 0 and vertical[i-1] != 'v':
                return False
        else:
            if i > 0 and vertical[i-1] != '^':
                return False
    
    return True
